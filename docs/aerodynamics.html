 50    def C_na(self, body):
 51        """Normal-force coefficient contribution from a body-area change.
 52
 53        Inputs:
 54            body: Body section with area_l and area_0 attributes describing the local area change.
 55
 56        Outputs:
 57            float: Normal-force coefficient contribution due to the area change.
 58        """
 59        if getattr(body, 'area_l', None) is None or getattr(body, 'area_0', None) is None:
 60            return 0.0
 61
 62        delta_area = float(body.area_l - body.area_0)
 63        if np.isclose(delta_area, 0.0):
 64            return 0.0
 65
 66        if np.isclose(self.alpha, 0.0):
 67            return 2.0 * delta_area / self.Aref
 68        return 2.0 * delta_area * np.sin(self.alpha) / (self.alpha * self.Aref)

 70    def C_n_lift(self, body, K=1.1):
 71        """Lift-induced normal-force contribution for a cylindrical body section.
 72
 73        Inputs:
 74            body: Cylindrical body section with radius and length values.
 75            K: Empirical lift scaling factor, defaults to 1.1.
 76
 77        Outputs:
 78            float: Normal-force coefficient contribution from lift-induced effects.
 79        """
 80        if getattr(body, 'radius', None) is None or getattr(body, 'length', None) is None:
 81            return 0.0
 82        if body.length <= 0 or body.radius <= 0:
 83            return 0.0
 84
 85        return K * 2.0 * body.radius * body.length * (np.sin(self.alpha) ** 2) / self.Aref

 87    def C_na_fins(self, fin, tube):
 88        """Normal-force coefficient contribution for a fin set, including tube effects.
 89
 90        Inputs:
 91            fin: Fin geometry object with span, area, and sweep-related properties.
 92            tube: Tube or body section used to apply a tube correction factor.
 93
 94        Outputs:
 95            float: Normal-force coefficient contribution for the fin set.
 96        """
 97        if getattr(fin, 'span', None) is None or getattr(fin, 'area', None) is None:
 98            return 0.0
 99
100        beta = np.sqrt(np.abs(1.0 - self.mach ** 2))
101        if self.mach <= 1.0:
102            denom = 1.0 + np.sqrt(1.0 + (beta * (fin.span ** 2) / (fin.area * np.cos(fin.Lc))) ** 2)
103            c_na_single = (2.0 * np.pi * (fin.span ** 2) / self.Aref) / denom
104        else:
105            k1 = 2.0 / beta
106            k2 = ((self.gamma + 1.0) * self.mach ** 4 - 4.0 * beta ** 2) / (4.0 * beta ** 4)
107            k3 = ((self.gamma + 1.0) * self.mach ** 8 + (2.0 * self.gamma ** 2 - 7.0 * self.gamma - 5.0) * self.mach ** 6 + 10.0 * (self.gamma + 1.0) * self.mach ** 4 + 8.0) / (6.0 * beta ** 7)
108            c_na_single = (fin.area / self.Aref) * (k1 + k2 * self.alpha + k3 * self.alpha ** 2)
109
110        c_na = 2.0 * c_na_single
111        if getattr(tube, 'radius', None) is not None and getattr(tube, 'span', None) is not None:
112            kt = 1.0 + tube.radius / (tube.span + tube.radius)
113            c_na *= kt
114        return c_na

116    def C_n_total(self, rocket):
117        """Sum the normal-force contributions from all body parts and fins.
118
119        Inputs:
120            rocket: Rocket configuration object containing parts, fins, and fin_tube data.
121
122        Outputs:
123            float: Total normal-force coefficient for the rocket configuration.
124        """
125        cn_total = 0.0
126        for part in getattr(rocket, 'parts', []):
127            area_change = getattr(part, 'area_l', None) is not None and getattr(part, 'area_0', None) is not None
128            if area_change and not np.isclose(part.area_l, part.area_0):
129                cn_total += self.C_na(part) * self.alpha
130
131            if getattr(part, 'radius', None) is not None and getattr(part, 'length', None) is not None:
132                if part.length > 0 and part.radius > 0:
133                    k_lift = 1.1
134                    if area_change and np.isclose(part.area_l, part.area_0):
135                        k_lift = 0.6 #TODO where did AI get this
136                    cn_total += self.C_n_lift(part, K=k_lift)
137
138        if getattr(rocket, 'fins', None) is not None and getattr(rocket, 'fin_tube', None) is not None:
139            cn_total += self.C_na_fins(rocket.fins, rocket.fin_tube) * self.alpha
140        return cn_total

144    def C_ma(self, body):
145        """Moment coefficient due to a body-area change about the reference axis.
146
147        Inputs:
148            body: Body section with area change, radius, length, and volume attributes.
149
150        Outputs:
151            float: Pitching-moment coefficient contribution caused by the area change.
152        """
153        if getattr(body, 'area_l', None) is None or getattr(body, 'area_0', None) is None:
154            return 0.0
155
156        delta_area = float(body.area_l - body.area_0)
157        if np.isclose(delta_area, 0.0):
158            return 0.0
159
160        numerator = body.length * body.area_l - body.volume
161        if np.isclose(self.alpha, 0.0):
162            return numerator / (self.Aref * body.radius)
163        return numerator * np.sin(self.alpha) / (self.alpha * self.Aref * body.radius)

264    def cp_fins(self, fins):
265        """Return the chordwise and spanwise centre-of-pressure coordinates for fins.
266
267        Inputs:
268            fins: Fin geometry object containing span, root_chord, tip_chord, sweep_length, AR, and c_hat values.
269
270        Outputs:
271            list: [x, y] centre-of-pressure coordinates in chordwise and spanwise directions.
272        """
273        y = (fins.span / 3.0) * (fins.root_chord + 2.0 * fins.tip_chord) / (fins.root_chord + fins.tip_chord)
274        if self.mach < 1.0:
275            x = (fins.sweep_length / 3.0) * (fins.root_chord + 2.0 * fins.tip_chord) / (fins.root_chord + fins.tip_chord) + (fins.root_chord ** 2 + fins.root_chord * fins.tip_chord + 3.0 * fins.tip_chord ** 2) / (6.0 * (fins.root_chord + fins.tip_chord))
276        elif self.mach < 1.3:
277            x_sub = (fins.sweep_length / 3.0) * (fins.root_chord + 2.0 * fins.tip_chord) / (fins.root_chord + fins.tip_chord) + (fins.root_chord ** 2 + fins.root_chord * fins.tip_chord + 3.0 * fins.tip_chord ** 2) / (6.0 * (fins.root_chord + fins.tip_chord))
278            beta = np.sqrt(np.abs(1.0 - self.mach ** 2))
279            x_sup = (fins.ar * beta - 0.67) / (2.0 * fins.ar * beta - 1.0) * fins.c_hat
280            t = (self.mach - 1.0) / 0.3
281            x = x_sub + t * (x_sup - x_sub)
282        else:
283            beta = np.sqrt(np.abs(1.0 - self.mach ** 2))
284            x = (fins.ar * beta - 0.67) / (2.0 * fins.ar * beta - 1.0) * fins.c_hat
285        return [x, y]

321    def skin_friction(self, rocket):
322        """Estimate skin-friction drag for the rocket body and fins.
323
324        Inputs:
325            rocket: Rocket configuration with body parts, roughness, and fin geometry.
326
327        Outputs:
328            float: Skin-friction drag coefficient contribution for the rocket.
329        """
330        velocity = self.mach * np.sqrt(self.gamma * self.R_const * self.temp)
331        cd_skin = 0.0
332
333        for part in getattr(rocket, 'parts', []):
334            if getattr(part, 'length', None) is None:
335                continue
336            L = max(float(part.length), 1e-9)
337            Re = velocity * L / self.kv
338            roughness_ratio = getattr(rocket, 'roughness', 0.0) / L if L > 0 else 0.0
339            Re_crit = 51.0 * (roughness_ratio) ** -1.039 if roughness_ratio > 0 else 1e9
340
341            if Re < 1e-4:
342                cf = 1.48e-2
343            elif Re < Re_crit:
344                cf = 1.0 / ((1.5 * np.log(Re) - 5.6) ** 2)
345            else:
346                cf = 0.032 * (roughness_ratio ** 0.2)
347
348            if self.mach < 1.0:
349                cf *= (1.0 - 0.1 * self.mach ** 2)
350            elif Re < Re_crit:
351                cf /= (1.0 + 0.15 * self.mach ** 2) ** 0.58
352            else:
353                cf /= (1.0 + 0.18 * self.mach ** 2)
354
355            area = getattr(part, 'Awet', None)
356            if area is None and getattr(part, 'radius', None) is not None:
357                area = 2.0 * np.pi * part.radius * L
358            if area is not None:
359                cd_skin += (cf * area / self.Aref)
360
361        if getattr(rocket, 'fins', None) is not None:
362            fin = rocket.fins
363            span = max(float(fin.span), 1e-9)
364            fin_Re = velocity * span / self.kv
365            roughness_ratio = getattr(rocket, 'roughness', 0.0) / span if span > 0 else 0.0
366            Re_crit_fin = 51.0 * (roughness_ratio) ** -1.039 if roughness_ratio > 0 else 1e9
367            if fin_Re < 1e-4:
368                fin_cf = 1.48e-2
369            elif fin_Re < Re_crit_fin:
370                fin_cf = 1.0 / ((1.5 * np.log(fin_Re) - 5.6) ** 2)
371            else:
372                fin_cf = 0.032 * (roughness_ratio ** 0.2)
373            if self.mach < 1.0:
374                fin_cf *= (1.0 - 0.1 * self.mach ** 2)
375            cd_skin += (fin_cf / self.Aref) * (1.0 + 2.0 * fin.thickness / fin.c_hat) * 8.0 * fin.area
376
377        return cd_skin

418    def cd_ogive(self, nosecone):
419        """Drag coefficient for an ogive nosecone over the current Mach range.
420
421        Inputs:
422            nosecone: Ogive nosecone object containing shape_factor, shoulder_angle, half_angle, and related geometry.
423
424        Outputs:
425            float: Drag coefficient for the ogive nosecone.
426        """
427        if self.mach > 1.0:
428            return (0.72 * (nosecone.shape_factor - 0.5) ** 2 + 0.82) * self.cd_cone(nosecone)
429
430        cd_m0 = 0.8 * (np.sin(nosecone.shoulder_angle) ** 2)
431        cd_m1 = (0.72 * (nosecone.shape_factor - 0.5) ** 2 + 0.82) * np.sin(nosecone.half_angle)
432        cd_m1_slope = 4.0 * (1.0 - 0.5 * np.sin(nosecone.half_angle)) / (self.gamma + 1.0)
433
434        b = cd_m1_slope / (cd_m1 - cd_m0)
435        a = cd_m1 - cd_m0
436        return a * np.power(self.mach, b) + cd_m0

438    def cd_cone(self, nosecone):
439        """Drag coefficient for a conical nose section as a function of Mach.
440
441        Inputs:
442            nosecone: Conical nose section with half_angle geometry.
443
444        Outputs:
445            float: Drag coefficient for the conical nose section.
446        """
447        if self.mach >= 1.3:
448            return 2.1 * (np.sin(nosecone.half_angle) ** 2) + 0.5 * np.sin(nosecone.half_angle) / np.sqrt((self.mach ** 2) - 1.0)
449
450        cd1 = np.sin(nosecone.half_angle)
451        cd2 = 2.1 * (np.sin(nosecone.half_angle) ** 2) + 0.5 * np.sin(nosecone.half_angle) / np.sqrt((1.3 ** 2) - 1.0)
452        cd1_slope = 4.0 * (1.0 - 0.5 * np.sin(nosecone.half_angle)) / (self.gamma + 1.0)
453
454        a = (-10.0 / 9.0) * (10.0 * cd1 - 10.0 * cd2 + 3.0 * cd1_slope)
455        b = (200.0 / 9.0) * (cd1 - cd2) + (23.0 * cd1_slope / 3.0)
456        c = (1.0 / 9.0) * (-91.0 * cd1 + 100.0 * cd2 - 39.0 * cd1_slope)

 51    def C_na(self, body):
 52        """Normal-force coefficient contribution from a body-area change.
 53
 54        Inputs:
 55            body: Body section with area_l and area_0 attributes describing the local area change.
 56
 57        Outputs:
 58            float: Normal-force coefficient contribution due to the area change.
 59        """
 60        if getattr(body, 'area_l', None) is None or getattr(body, 'area_0', None) is None:
 61            return 0.0
 62
 63        delta_area = float(body.area_l - body.area_0)
 64        if np.isclose(delta_area, 0.0):
 65            return 0.0
 66
 67        if np.isclose(self.alpha, 0.0):
 68            return 2.0 * delta_area / self.Aref
 69        return 2.0 * delta_area * np.sin(self.alpha) / (self.alpha * self.Aref)

 71    def C_n_lift(self, body, K=1.1):
 72        """Lift-induced normal-force contribution for a cylindrical body section.
 73
 74        Inputs:
 75            body: Cylindrical body section with radius and length values.
 76            K: Empirical lift scaling factor, defaults to 1.1.
 77
 78        Outputs:
 79            float: Normal-force coefficient contribution from lift-induced effects.
 80        """
 81        if getattr(body, 'radius', None) is None or getattr(body, 'length', None) is None:
 82            return 0.0
 83        if body.length <= 0 or body.radius <= 0:
 84            return 0.0
 85
 86        return K * 2.0 * body.radius * body.length * (np.sin(self.alpha) ** 2) / self.Aref

 88    def C_na_fins(self, fin, tube):
 89        """Normal-force coefficient contribution for a fin set, including tube effects.
 90
 91        Inputs:
 92            fin: Fin geometry object with span, area, and sweep-related properties.
 93            tube: Tube or body section used to apply a tube correction factor.
 94
 95        Outputs:
 96            float: Normal-force coefficient contribution for the fin set.
 97        """
 98        if getattr(fin, 'span', None) is None or getattr(fin, 'area', None) is None:
 99            return 0.0
100
101        beta = np.sqrt(np.abs(1.0 - self.mach ** 2))
102        if self.mach <= 1.0:
103            denom = 1.0 + np.sqrt(1.0 + (beta * (fin.span ** 2) / (fin.area * np.cos(fin.Lc))) ** 2)
104            c_na_single = (2.0 * np.pi * (fin.span ** 2) / self.Aref) / denom
105        else:
106            k1 = 2.0 / beta
107            k2 = ((self.gamma + 1.0) * self.mach ** 4 - 4.0 * beta ** 2) / (4.0 * beta ** 4)
108            k3 = ((self.gamma + 1.0) * self.mach ** 8 + (2.0 * self.gamma ** 2 - 7.0 * self.gamma - 5.0) * self.mach ** 6 + 10.0 * (self.gamma + 1.0) * self.mach ** 4 + 8.0) / (6.0 * beta ** 7)
109            c_na_single = (fin.area / self.Aref) * (k1 + k2 * self.alpha + k3 * self.alpha ** 2)
110
111        c_na = 2.0 * c_na_single
112        if getattr(tube, 'radius', None) is not None and getattr(tube, 'span', None) is not None:
113            kt = 1.0 + tube.radius / (tube.span + tube.radius)
114            c_na *= kt
115        return c_na

323        """Estimate skin-friction drag for the rocket body and fins.
324
325        Inputs:
326            rocket: Rocket configuration with body parts, roughness, and fin geometry.
327
328        Outputs:
329            float: Skin-friction drag coefficient contribution for the rocket.
330        """
331        velocity = self.mach * np.sqrt(self.gamma * self.R_const * self.temp)
332        cd_skin = 0.0
333
334        for part in getattr(rocket, 'parts', []):
335            if getattr(part, 'length', None) is None:
336                continue
337            L = max(float(part.length), 1e-9)
338            Re = velocity * L / self.kv
339            roughness_ratio = getattr(rocket, 'roughness', 0.0) / L if L > 0 else 0.0
340            Re_crit = 51.0 * (roughness_ratio) ** -1.039 if roughness_ratio > 0 else 1e9
341
342            if Re < 1e-4:
343                cf = 1.48e-2
344            elif Re < Re_crit:
345                cf = 1.0 / ((1.5 * np.log(Re) - 5.6) ** 2)
346            else:
347                cf = 0.032 * (roughness_ratio ** 0.2)
348
349            if self.mach < 1.0:
350                cf *= (1.0 - 0.1 * self.mach ** 2)
351            elif Re < Re_crit:
352                cf /= (1.0 + 0.15 * self.mach ** 2) ** 0.58
353            else:
354                cf /= (1.0 + 0.18 * self.mach ** 2)
355
356            area = getattr(part, 'Awet', None)
357            if area is None and getattr(part, 'radius', None) is not None:
358                area = 2.0 * np.pi * part.radius * L
359            if area is not None:
360                cd_skin += (cf * area / self.Aref)
361
362        if getattr(rocket, 'fins', None) is not None:
363            fin = rocket.fins
364            span = max(float(fin.span), 1e-9)
